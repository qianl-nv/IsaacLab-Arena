cache = Path.home() / ".cache/huggingface/hub/datasets--nvidia--simready-assets/snapshots/653d91e182011f357cf1644013cc94ddf194d349"
for asset in ("furniture_workbench_standard_e01", "fixture_cabinet_b01"):
